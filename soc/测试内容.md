menuentry "启动 Kali Linux (自动BIOS/UEFI)" --class=linux {

    # 自动判断启动模式

    if [ "$grub_platform" = "pc" ]; then

        # BIOS Legacy 模式

        insmod part_msdos

    else

        # UEFI 模式

        insmod part_gpt

    fi

    insmod ext2

    set root=($vtoydev,3)

    # 内核文件在sda3根目录，不是boot文件夹

    linux /vmlinuz root=UUID=EFD85ADA-77BF-4A54-AFDA-565C9C6DC8E0 rw

    initrd /initrd.img

}
